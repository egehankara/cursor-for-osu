cmake_minimum_required(VERSION 3.22)
project(osu-cursor-trail LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(ECM REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH "${ECM_MODULE_PATH}")
set(QT_MAJOR_VERSION 6)
include(KDEInstallDirs6)
find_package(Qt6 REQUIRED COMPONENTS DBus Widgets)
find_package(KWin 6.6 REQUIRED)

add_library(osu-cursor-trail MODULE trail.cpp trailstate.cpp)
target_link_libraries(osu-cursor-trail PRIVATE KWin::kwin)
set_target_properties(osu-cursor-trail PROPERTIES PREFIX "")
set_property(TARGET osu-cursor-trail APPEND PROPERTY AUTOMOC_MACRO_NAMES KWIN_EFFECT_FACTORY_SUPPORTED)
install(TARGETS osu-cursor-trail DESTINATION "${KDE_INSTALL_PLUGINDIR}/kwin/effects/plugins")

include(CTest)
if(BUILD_TESTING)
    add_executable(trail-test trailtest.cpp trailstate.cpp)
    target_link_libraries(trail-test PRIVATE Qt6::Gui)
    add_test(NAME trail-state-and-expiration COMMAND trail-test)
endif()
